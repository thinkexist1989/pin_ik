cmake_minimum_required(VERSION 3.16)
project(ik_comparison LANGUAGES CXX)
set(IK_BACKEND "PIN" CACHE STRING "PIN or KDL")
set(IK_SOURCE_DIR "" CACHE PATH "Solver source checkout")
set(IK_BUILD_DIR "" CACHE PATH "Built solver directory")
find_package(Eigen3 REQUIRED)
find_package(NLopt REQUIRED)
find_package(Threads REQUIRED)
add_executable(ik_benchmark ik_benchmark.cpp)
target_compile_features(ik_benchmark PRIVATE cxx_std_17)
target_link_libraries(ik_benchmark PRIVATE Eigen3::Eigen ${NLOPT_LIBRARIES} Threads::Threads)
if(IK_BACKEND STREQUAL "KDL")
  find_package(orocos_kdl REQUIRED)
  target_compile_definitions(ik_benchmark PRIVATE BENCH_KDL)
  target_include_directories(ik_benchmark PRIVATE ${IK_SOURCE_DIR}/trac_ik_lib/include ${orocos_kdl_INCLUDE_DIRS})
  find_library(IK_LIBRARY trac_ik PATHS ${IK_BUILD_DIR}/trac_ik_lib NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(ik_benchmark PRIVATE ${IK_LIBRARY} ${orocos_kdl_LIBRARIES})
elseif(IK_BACKEND STREQUAL "PIN")
  find_package(pinocchio REQUIRED)
  target_include_directories(ik_benchmark PRIVATE ${IK_SOURCE_DIR}/include)
  find_library(IK_LIBRARY pin_ik PATHS ${IK_BUILD_DIR} NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(ik_benchmark PRIVATE ${IK_LIBRARY} pinocchio::pinocchio)
else()
  message(FATAL_ERROR "IK_BACKEND must be PIN or KDL")
endif()
