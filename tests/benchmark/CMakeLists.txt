cmake_minimum_required(VERSION 3.16)
project(ik_comparison LANGUAGES CXX)
set(IK_BACKEND "PIN" CACHE STRING "PIN or KDL")
set(IK_SOURCE_DIR "" CACHE PATH "Solver source checkout")
set(IK_BUILD_DIR "" CACHE PATH "Built solver directory")
option(IK_ENABLE_THREAD_CONFIG_BENCHMARKS "Build Linux stage D configuration benchmarks (requires stage D library)" OFF)
find_package(Eigen3 REQUIRED)
find_package(NLopt REQUIRED)
find_package(Threads REQUIRED)
add_executable(ik_benchmark ik_benchmark.cpp)
target_compile_features(ik_benchmark PRIVATE cxx_std_17)
target_link_libraries(ik_benchmark PRIVATE Eigen3::Eigen ${NLOPT_LIBRARIES} Threads::Threads)
if(IK_BACKEND STREQUAL "KDL")
  find_package(orocos_kdl REQUIRED)
  target_compile_definitions(ik_benchmark PRIVATE BENCH_KDL)
  target_include_directories(ik_benchmark PRIVATE ${IK_SOURCE_DIR}/trac_ik_lib/include ${orocos_kdl_INCLUDE_DIRS})
  find_library(IK_LIBRARY trac_ik PATHS ${IK_BUILD_DIR}/trac_ik_lib NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(ik_benchmark PRIVATE ${IK_LIBRARY} ${orocos_kdl_LIBRARIES})
elseif(IK_BACKEND STREQUAL "PIN")
  find_package(pinocchio REQUIRED)
  target_include_directories(ik_benchmark PRIVATE ${IK_SOURCE_DIR}/include)
  find_library(IK_LIBRARY pin_ik PATHS ${IK_BUILD_DIR} NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(ik_benchmark PRIVATE ${IK_LIBRARY} pinocchio::pinocchio)
  add_executable(stage_c_probe stage_c_probe.cpp)
  target_compile_features(stage_c_probe PRIVATE cxx_std_17)
  # Diagnostic executable and its support library use the same test-enabled
  # class definitions. Formal timing remains on the production ik_benchmark.
  target_compile_definitions(stage_c_probe PRIVATE PIN_IK_TESTING)
  target_include_directories(stage_c_probe PRIVATE ${IK_SOURCE_DIR}/include)
  target_link_libraries(stage_c_probe PRIVATE ${IK_BUILD_DIR}/tests/libpin_ik_worker_test_support.a
    pinocchio::pinocchio ${NLOPT_LIBRARIES} Threads::Threads)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    if(IK_ENABLE_THREAD_CONFIG_BENCHMARKS)
      add_executable(stage_d_benchmark ik_benchmark.cpp)
      target_compile_features(stage_d_benchmark PRIVATE cxx_std_17)
      target_compile_definitions(stage_d_benchmark PRIVATE BENCH_WORKER_CONFIG)
      target_include_directories(stage_d_benchmark PRIVATE ${IK_SOURCE_DIR}/include)
      target_link_libraries(stage_d_benchmark PRIVATE ${IK_LIBRARY} pinocchio::pinocchio ${NLOPT_LIBRARIES} Threads::Threads)
      add_executable(stage_d_timing_probe ik_benchmark.cpp)
      target_compile_features(stage_d_timing_probe PRIVATE cxx_std_17)
      target_compile_definitions(stage_d_timing_probe PRIVATE PIN_IK_TESTING BENCH_TIMING BENCH_WORKER_CONFIG)
      target_include_directories(stage_d_timing_probe PRIVATE ${IK_SOURCE_DIR}/include)
      target_link_libraries(stage_d_timing_probe PRIVATE ${IK_BUILD_DIR}/tests/libpin_ik_worker_test_support.a
        pinocchio::pinocchio ${NLOPT_LIBRARIES} Threads::Threads)
    endif()
    add_executable(timing_probe ik_benchmark.cpp)
    target_compile_features(timing_probe PRIVATE cxx_std_17)
    target_compile_definitions(timing_probe PRIVATE PIN_IK_TESTING BENCH_TIMING)
    target_include_directories(timing_probe PRIVATE ${IK_SOURCE_DIR}/include)
    target_link_libraries(timing_probe PRIVATE ${IK_BUILD_DIR}/tests/libpin_ik_worker_test_support.a
      pinocchio::pinocchio ${NLOPT_LIBRARIES} Threads::Threads)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(allocation_counter SHARED allocation_counter.cpp)
    add_executable(allocation_probe allocation_probe.cpp)
    target_compile_features(allocation_probe PRIVATE cxx_std_17)
    target_include_directories(allocation_probe PRIVATE ${IK_SOURCE_DIR}/include)
    target_link_libraries(allocation_probe PRIVATE ${IK_LIBRARY} pinocchio::pinocchio ${NLOPT_LIBRARIES} Threads::Threads ${CMAKE_DL_LIBS})
  endif()
else()
  message(FATAL_ERROR "IK_BACKEND must be PIN or KDL")
endif()
